# Install and load the Owl Post launchd job. Run these yourself, one at a time.
# With terminal loops (~/.hogwarts/loops/jobs is there) skip this file: run
# scripts/owlpost-setup.sh, which adds owlpost to that list and loads nothing here.

# 1. The castle outboxes must exist first, because launchd watches them.
ls -ld /Users/crisryantan/hogwarts/desks/*/outbox

# 2. The office logs folder, mode 0700.
mkdir -m 700 -p /Users/crisryantan/.hogwarts/logs

# 3. Run one pass by hand. It should print {"ok": true, ...} with nothing delivered.
/usr/bin/env -i /usr/bin/python3 -I -B -X pycache_prefix=/var/empty -c 'import sys; sys.path.insert(0, "/Users/crisryantan/.hogwarts"); from fleet.owl_post import main; sys.exit(main())'

# 4. Check the template, then copy it in.
plutil -lint /Users/crisryantan/.hogwarts/launchd/com.hogwarts.owlpost.plist
cp /Users/crisryantan/.hogwarts/launchd/com.hogwarts.owlpost.plist ~/Library/LaunchAgents/com.hogwarts.owlpost.plist
chmod 644 ~/Library/LaunchAgents/com.hogwarts.owlpost.plist

# 5. Load it and look at it.
launchctl bootstrap gui/$(id -u) ~/Library/LaunchAgents/com.hogwarts.owlpost.plist
launchctl print gui/$(id -u)/com.hogwarts.owlpost | head -30

# 6. Test: write one owl and watch it move.
#    /Users/crisryantan/hogwarts/desks/mcgonagall/outbox/hello.json containing
#    {"to": "hermione", "kind": "fyi", "subject": "owl post test", "body": "hello"}
#    Then check desks/hermione/inbox/ and desks/mcgonagall/outbox/.sent/
tail -5 /Users/crisryantan/.hogwarts/logs/owlpost.out.log /Users/crisryantan/.hogwarts/logs/owlpost.err.log

# Undo.
launchctl bootout gui/$(id -u)/com.hogwarts.owlpost
rm ~/Library/LaunchAgents/com.hogwarts.owlpost.plist
